២. កូដ VBA Macro សម្រាប់បង្កើត Function =CHHANKITEK(A2)

លោកអ្នកអាចចុច Alt + F11 ក្នុង Excel រួច Insert Module និងបិទភ្ជាប់ (Paste) កូដខាងក្រោម ដើម្បីប្រើប្រាស់អនុគមន៍ផ្ទាល់ខ្លួន =CHHANKITEK(ក្រឡាថ្ងៃខែ) បានគ្រប់ទីកន្លែង៖ 

' Khmer Lunar Calendar Function for Excel VBA
Public Function CHHANKITEK(dInput As Variant) As String
    On Error GoTo ErrHandler
    Dim wsDB As Worksheet
    Set wsDB = ThisWorkbook.Sheets("ទិន្នន័យចន្ទគតិ_DB")
    Dim r As Range, matchRow As Long
    Set r = wsDB.Range("A:A")
    matchRow = Application.WorksheetFunction.Match(CLng(CDate(dInput)), r, 0)
    CHHANKITEK = wsDB.Cells(matchRow, 5).Value
    Exit Function
ErrHandler:
    CHHANKITEK = "កាលបរិច្ឆេទមិនត្រឹមត្រូវ"
End Function
